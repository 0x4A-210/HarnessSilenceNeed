000021: // When this quirk is set, a positive or zero value ignores or enforces the "If

000025: pub const QUIRK_ALLOW_NON_ZERO_INITIAL_BYTE : base.u32 = 0x5058_E000 | 0x00
000026: 
000027: // When this quirk is set, a positive value indicates an extension to the LZMA
000028: // format. Zero means to use the default LZMA format.
000029: //
000030: // When positive, the low byte of the u64 value indicates the extension.
000031: // Currently, the only valid extensions are:
000032: //   - 0x01 indicates Lzip.
000033: //   - 0x02 indicates LZMA2.
000034: //
000035: // For Lzip, the high 7 bytes (shifted right by 8) must be less than or equal
000036: // to 0xFF and indicates the dictionary size. See section "6 File format" of
000037: // https://www.nongnu.org/lzip/manual/lzip_manual.html
000038: //
000039: // For LZMA2, the high 7 bytes (shifted right by 8) must be less than or equal
000040: // to 40 and indicates the dictionary size. See section "5.3.1. LZMA2" of
000041: // https://tukaani.org/xz/xz-file-format.txt
000042: pub const QUIRK_FORMAT_EXTENSION : base.u32 = 0x5058_E000 | 0x01
